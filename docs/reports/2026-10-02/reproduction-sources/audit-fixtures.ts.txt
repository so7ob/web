import { PrismaClient } from '@prisma/client';
import bcrypt from 'bcryptjs';
import { randomBytes } from 'crypto';
import { writeFileSync, mkdirSync } from 'fs';
const db = new PrismaClient();
const password = randomBytes(24).toString('hex');
const passwordHash = await bcrypt.hash(password,12);
const roles = {admin:'super_admin',ops:'ops_manager',support:'support',editor:'content_editor',owner:'client',other:'client',pending:'client',suspended:'client'};
for (const [id,roleKey] of Object.entries(roles)) await db.user.create({data:{id,email:`${id}@audit.example`,name:`Synthetic ${id}`,passwordHash,roleKey,status:id==='suspended'?'suspended':id==='pending'?'pending_verification':'active',emailVerifiedAt:id==='pending'?null:new Date()}});
const req = await db.projectRequest.create({data:{id:'auditrequest',refCode:'S7-AUDIT000',requestType:'discussion',serviceType:'web',description:'Synthetic request for local audit only',budget:'unspecified',timeline:'flexible',name:'Synthetic owner',email:'owner@audit.example',preferredContact:'email',descriptionHash:'synthetic',clientId:'owner',assigneeId:'support'}});
await db.requestMessage.createMany({data:[{requestId:req.id,authorId:'support',authorType:'staff',kind:'internal_note',body:'AUDIT_INTERNAL_MARKER'},{requestId:req.id,authorId:'support',authorType:'staff',kind:'message',body:'Synthetic visible message'}]});
await db.inquiry.create({data:{id:'auditinquiry',refCode:'IQ-AUDIT000',subject:'Synthetic inquiry',category:'general',name:'Synthetic owner',email:'owner@audit.example',clientId:'owner',assigneeId:'support',messages:{create:{authorId:'support',authorType:'staff',kind:'internal_note',body:'AUDIT_INTERNAL_MARKER'}}}});
mkdirSync('db/uploads',{recursive:true});
writeFileSync('db/uploads/audit.txt','synthetic private attachment');
await db.attachment.createMany({data:[{id:'requestattachment',filename:'audit.txt',storedName:'audit.txt',mimeType:'text/plain',size:28,uploaderId:'owner',requestId:'auditrequest'},{id:'inquiryattachment',filename:'inquiry.txt',storedName:'inquiry-audit.txt',mimeType:'text/plain',size:28,uploaderId:'owner',inquiryId:'auditinquiry'}]});
writeFileSync('db/uploads/inquiry-audit.txt','synthetic private attachment');
writeFileSync('.audit-secrets.json',JSON.stringify({password}),{mode:0o600});
console.log(JSON.stringify({syntheticUsers:8,requests:1,inquiries:1,attachments:2}));
await db.$disconnect();
