const {chromium}=require('/usr/lib/node_modules/playwright');const fs=require('fs');
(async()=>{const browser=await chromium.launch({headless:true}),rows=[];for(const width of [375,1280])for(const locale of ['ar','en']){const ctx=await browser.newContext({viewport:{width,height:900},reducedMotion:'no-preference'});await ctx.route('**/*',r=>new URL(r.request().url()).hostname==='localhost'?r.continue():r.abort());await ctx.addInitScript(()=>{window.__m={lcp:0,cls:0};new PerformanceObserver(l=>{for(const e of l.getEntries())window.__m.lcp=e.startTime}).observe({type:'largest-contentful-paint',buffered:true});new PerformanceObserver(l=>{for(const e of l.getEntries())if(!e.hadRecentInput)window.__m.cls+=e.value}).observe({type:'layout-shift',buffered:true})});const page=await ctx.newPage();await page.goto('http://localhost:3100/'+locale,{waitUntil:'networkidle'});await page.waitForTimeout(1500);rows.push(await page.evaluate(({locale,width})=>({locale,width,...window.__m,heroOpacity:getComputedStyle(document.querySelector('h1').parentElement).opacity}),{locale,width}));await ctx.close()}await browser.close();fs.writeFileSync('/tmp/so7ob-audit-20261002/artifacts/browser-home-normal.json',JSON.stringify(rows,null,2));console.log(JSON.stringify(rows))})();
